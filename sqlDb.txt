-- Table Utilisateur
CREATE TABLE Utilisateur (
    idUtilisateur NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nom VARCHAR2(255) NOT NULL,
    prenom VARCHAR2(255) NOT NULL,
    email VARCHAR2(255) UNIQUE NOT NULL,
    motDePasse VARCHAR2(255) NOT NULL,
    adresse VARCHAR2(255),
    dateInscription DATE DEFAULT SYSDATE
);

-- Table Document
CREATE TABLE Documents (
    idDocument NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    titre VARCHAR2(255) NOT NULL,
    auteur VARCHAR2(255),
    isbn VARCHAR2(20) UNIQUE,
    categorie VARCHAR2(255),
    nombreExemplaires NUMBER DEFAULT 0,
    disponible NUMBER(1, 0) DEFAULT 1 -- 1 pour disponible, 0 pour non disponible
);

-- Table Reservation
CREATE TABLE Reservation (
    idReservation NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    idUtilisateur NUMBER,
    idDocument NUMBER,
    dateReservation DATE DEFAULT SYSDATE,
    dateExpiration DATE,
    statut VARCHAR2(50),
    FOREIGN KEY (idUtilisateur) REFERENCES Utilisateur(idUtilisateur),
    FOREIGN KEY (idDocument) REFERENCES Documents(idDocument)
);

-- Table Emprunt
CREATE TABLE Emprunt (
    idEmprunt NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    idUtilisateur NUMBER,
    idDocument NUMBER,
    dateEmprunt DATE DEFAULT SYSDATE,
    dateRetourPrevue DATE,
    dateRetourEffective DATE,
    enRetard NUMBER(1, 0) DEFAULT 0, -- 1 pour en retard, 0 pour à temps
    FOREIGN KEY (idUtilisateur) REFERENCES Utilisateur(idUtilisateur),
    FOREIGN KEY (idDocument) REFERENCES Documents(idDocument)
);
